   string temperature = "5 6 7 -18 5 4 -4 -18";

   List<string> list = temperature.Split(' ').ToList();

   List<int> ints = new List<int>();

   foreach (string s in list)
   {
       int x = int.Parse(s);
       ints.Add(x);
   }

   foreach (int x in ints)
   {
       Console.WriteLine(x);
   }

   Console.WriteLine(ints.Count);

   List<int> negativTemp = new List<int>();

   foreach (int x in ints)
   {
       if (x < 0)
       {
           if (!negativTemp.Contains(x))
           {
               negativTemp.Add(x);
           }
          
       }
   }

   Console.WriteLine(string.Join(", ", negativTemp));
